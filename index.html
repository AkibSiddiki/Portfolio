public function show(string $slug): JsonResponse
{
$cacheKey = "article:slug:{$slug}";
$article = Cache::remember($cacheKey, 3600, function () use ($slug) {
return Article::where('slug', $slug)
->with(['category', 'tags', 'author'])
->published()
->firstOrFail();
});
return response()->json($article);
}

public function calculateMonthlySalary(Employee $emp): Payslip
{
$attendance = $emp->attendances()->inCurrentBillingCycle();
$gross = $emp->structure->basic + $emp->structure->allowances;
$deductions = TaxEngine::compute($gross, $emp->tier);
return Payslip::create([
'employee_id' => $emp->id,
'net_payable' => $gross - $deductions - $attendance->unpaidLoss(),
]);
}
